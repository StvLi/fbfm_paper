\newcommand{\UnifiedEndpointChain}{%
  \begin{aligned}
    \hat{\mathbf Q}_{t,k}^{\,1}

    \!\left(
      \mathbf Q_t^{\tau_k^Q},\tau_k^Q;
      \mathcal K_{t,k}^Q
    \right),\\
    \mathbf e_{t,k}^Q
    &=
    \mathbf P^Q\mathbf W_{t,k}^Q

      \mathbf Y_{t,k}^Q-\hat{\mathbf Q}_{t,k}^{\,1}
    \right),\\
    v_{\mathrm{FBFM}}^Q
    &=
    v_{\theta_Q}^Q
    +\lambda_{\tau_k^Q}^Q
    \left(\mathbf J_{t,k}^Q\right)^{\mathsf T}
    \mathbf e_{t,k}^Q
  \end{aligned}}
